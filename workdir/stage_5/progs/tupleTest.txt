decl
    int result, num;
    int add(int a, int b);

    tuple student(str name, int rollno, int class, str div) s1, s2;
enddecl


int add(int a, int b){
    begin
        return a + b;
    end
}


int main(){
    decl
        int a, b;
        str msg;
        tuple book(str title, int year) b1;
    enddecl

    begin

        // Normal variable testing
        a = 10;
        b = 20;
        result = add(a, b);
        write(result);  // expect 30

        // ---- Tuple field assignments ----
        s1.name = "Alice";
        s1.rollno = 21;
        s1.class = 3;
        s1.div = "C";

        s2.name = "Bob";
        s2.rollno = 22;
        s2.class = 3;
        s2.div = "B";

        b1.title = "Compiler";
        b1.year = 2025;

        // ---- Print tuple fields ----
        write(s1.name);
        write(s1.rollno);
        write(s2.name);
        write(s2.rollno);
        write(b1.title);
        write(b1.year);

        // ---- Mix tuple fields in expressions ----
        num = s1.rollno + s2.rollno;
        write(num);  // expect 43

        // ---- Reassign tuple fields ----
        s1.rollno = 99;
        write(s1.rollno); // expect 99

        return 1;
    end
}
