type
List
{
int data;
List next;
}
endtype
decl
List head[5];
enddecl

int main()
{
  decl
    int num, x;
  enddecl

  begin
    x = initialize();
    
    x = 0;
    while(x < 5) do
        read(num);
        head[x] = alloc();
        head[x].data = num;
        head[x].next = null;
        x = x + 1;
    endwhile;

    head[0].next = head[4];

    x = 0;
    while(x < 5) do
        while(head[x] != null) do
            write(head[x].data);
            head[x] = head[x].next;
        endwhile;
        x = x + 1;
    endwhile;

    return 1;
  end
}
