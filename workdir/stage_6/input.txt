type
    Student{
        str name;
        int rollno;
        int age;
    }
endtype
decl
    Student st[3];
enddecl

int main()
{
  decl
    int num, x, stage;
    str stname;
  enddecl

  begin
    x = initialize();
    
    x = 0;
    while(x < 3) do
        read(stname);
        read(num);
        read(stage);
        st[x] = alloc();
        st[x].name = stname;
        st[x].rollno = num;
        st[x].age = stage;
        x = x + 1;
    endwhile;

    write(" ");
    write("enter:");
    read(num);

    x = 0;
    while(x < 3) do
        if(st[x].rollno == num) then
            write(st[x].name);
            write(st[x].age);
        endif;
        x = x + 1;
    endwhile;

    return 1;
  end
}
